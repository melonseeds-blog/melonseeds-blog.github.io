이미지 처리 도구 개요

Python에서 이미지를 다룰 때 가장 많이 쓰는 도구는 Pillow, OpenCV-Python, scikit-image, imageio 네 가지다. 모두 결국 이미지를 numpy.ndarray 형태로 다루기 때문에 도구 간 변환이 자유롭다.

도구주력 용도특징
Pillow (PIL)단순 입출력·기본 변환API가 직관적, 표준에 가까움
OpenCV (cv2)본격 영상 처리·비전빠르고 기능 방대, C++ API와 거의 동일
scikit-image학술·알고리즘 연구numpy 친화, 알고리즘 풍부, 느릴 수 있음
imageio다양한 포맷·비디오 I/OGIF·동영상 입출력 간편

설치는 다음 한 줄로 끝난다.

pip install pillow opencv-python scikit-image imageio matplotlib numpy
핵심은 numpy ndarray 어느 라이브러리로 읽었든 결국 ndarray로 변환해 다룰 수 있다. ndarray는 Python 이미지 처리의 공용어다.

Pillow — 단순 입출력과 기본 변환

Pillow는 Python Imaging Library(PIL)의 후속 프로젝트다. API가 깔끔하고 설치도 가벼워 단순 작업에 적합하다.

from PIL import Image

# 읽기
img = Image.open("sample.jpg")
print(img.size)        # (width, height)
print(img.mode)        # 'RGB', 'L', 'RGBA' 등
print(img.format)      # 'JPEG'

# 저장
img.save("output.png")

# 리사이즈
resized = img.resize((640, 480))

# 자르기 (left, upper, right, lower)
cropped = img.crop((100, 100, 400, 400))

# 회전
rotated = img.rotate(45, expand=True)

# 흑백 변환
gray = img.convert("L")

# 좌우 반전
flipped = img.transpose(Image.FLIP_LEFT_RIGHT)

Pillow와 numpy 간 변환

import numpy as np
from PIL import Image

# Pillow → numpy
img = Image.open("sample.jpg")
arr = np.array(img)        # shape: (H, W, 3), RGB 순서, uint8

# numpy → Pillow
arr2 = np.zeros((480, 640, 3), dtype=np.uint8)
arr2[:, :, 0] = 255         # 빨간 이미지
img2 = Image.fromarray(arr2)
img2.save("red.png")

OpenCV-Python — 실전 처리

C++ OpenCV와 거의 동일한 API를 Python 바인딩으로 제공한다. 함수 이름·인자 순서가 같아 C++ 코드와 1:1로 옮기기 쉽다.

import cv2
import numpy as np

# 읽기 (BGR 순서로 반환됨)
img = cv2.imread("sample.jpg")           # 컬러
gray = cv2.imread("sample.jpg", cv2.IMREAD_GRAYSCALE)
print(img.shape)                          # (H, W, 3)
print(img.dtype)                          # uint8

# 저장
cv2.imwrite("output.png", img)

# 화면 표시 (스크립트에서만, Jupyter는 안 됨!)
cv2.imshow("preview", img)
cv2.waitKey(0)
cv2.destroyAllWindows()

# 색공간 변환
rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# 리사이즈
small = cv2.resize(img, (320, 240))
small2 = cv2.resize(img, None, fx=0.5, fy=0.5)

필터·임계값·모폴로지

# 가우시안 블러
blur = cv2.GaussianBlur(img, (5, 5), sigmaX=1.0)

# 메디안 블러 (소금후추 노이즈에 강함)
median = cv2.medianBlur(img, 5)

# 캐니 엣지
edges = cv2.Canny(gray, 100, 200)

# 이진화
_, binary = cv2.threshold(gray, 127, 255, cv2.THRESH_BINARY)
_, otsu = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)

# 모폴로지 (열기 = 침식 → 팽창, 잡음 제거)
kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (3, 3))
opened = cv2.morphologyEx(binary, cv2.MORPH_OPEN, kernel)

컨투어 추출과 그리기

# 외곽선 찾기
contours, hierarchy = cv2.findContours(
    binary, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE
)

# 모든 컨투어 그리기 (녹색, 두께 2)
result = img.copy()
cv2.drawContours(result, contours, -1, (0, 255, 0), 2)

# 각 컨투어의 면적·중심
for cnt in contours:
    area = cv2.contourArea(cnt)
    M = cv2.moments(cnt)
    if M["m00"] > 0:
        cx = int(M["m10"] / M["m00"])
        cy = int(M["m01"] / M["m00"])
        cv2.circle(result, (cx, cy), 4, (0, 0, 255), -1)

cv2와 numpy ndarray

OpenCV-Python의 이미지는 numpy ndarray다. 따라서 numpy의 슬라이싱·벡터 연산을 그대로 쓸 수 있다.

cv2.imread가 반환하는 ndarray의 구조 — 채널은 BGR 순서다.
img = cv2.imread("sample.jpg")
print(img.shape)    # (480, 640, 3) — (H, W, C)
print(img.dtype)    # uint8

# 픽셀 접근 (행=y, 열=x)
px = img[100, 200]              # [B, G, R]
b, g, r = img[100, 200]

# ROI 슬라이싱
roi = img[100:300, 50:200]      # y 100~300, x 50~200

# 채널 분리
b, g, r = cv2.split(img)
# 또는
b = img[:, :, 0]
g = img[:, :, 1]
r = img[:, :, 2]

# 채널 합치기
merged = cv2.merge([b, g, r])

# 벡터 연산 (모든 픽셀에 +50)
brighter = cv2.add(img, 50)     # 포화 연산 (255에서 멈춤)
# 주의: np.add나 + 연산자는 오버플로우 발생!

scikit-image — 알고리즘 라이브러리

scikit-image는 numpy 기반의 학술 친화적 이미지 처리 라이브러리다. 논문 코드와 잘 맞고 알고리즘이 풍부하지만 OpenCV보다 느린 편이다.

from skimage import io, filters, measure, morphology
import numpy as np

# 읽기 (RGB 순서, float64 또는 uint8)
img = io.imread("sample.jpg")
print(img.shape, img.dtype)

# 가우시안 블러
blurred = filters.gaussian(img, sigma=2.0, channel_axis=-1)

# Otsu 임계값
gray = img.mean(axis=2) if img.ndim == 3 else img
thr = filters.threshold_otsu(gray)
binary = gray > thr

# 모폴로지
cleaned = morphology.remove_small_objects(binary, min_size=64)

# 레이블링과 영역 분석
labels = measure.label(cleaned)
props = measure.regionprops(labels)
for p in props:
    print(p.area, p.centroid, p.bbox)

자주 쓰는 처리 흐름

대부분의 비전 작업은 다음 다섯 단계의 조합이다.

단계대표 함수목적
1. 읽기cv2.imread파일 → ndarray
2. 색공간 변환cv2.cvtColorBGR↔Gray↔HSV
3. 전처리GaussianBlur, threshold, morphologyEx잡음 제거·이진화
4. 분석findContours, connectedComponents객체 추출·측정
5. 저장·표시imwrite, imshow, plt.imshow결과 확인
import cv2
import numpy as np

# 1. 읽기
img = cv2.imread("part.jpg")

# 2. 색공간 변환
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# 3. 전처리: 블러 → Otsu 이진화 → 열기 연산
blur = cv2.GaussianBlur(gray, (5, 5), 0)
_, binary = cv2.threshold(blur, 0, 255,
                          cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)
kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (3, 3))
clean = cv2.morphologyEx(binary, cv2.MORPH_OPEN, kernel, iterations=2)

# 4. 분석: 컨투어 추출 → 면적·중심
contours, _ = cv2.findContours(clean, cv2.RETR_EXTERNAL,
                                cv2.CHAIN_APPROX_SIMPLE)
result = img.copy()
for cnt in contours:
    area = cv2.contourArea(cnt)
    if area < 100:
        continue
    x, y, w, h = cv2.boundingRect(cnt)
    cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)
    cv2.putText(result, f"{int(area)}", (x, y - 5),
                cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1)

# 5. 저장
cv2.imwrite("result.png", result)
print(f"detected: {len(contours)} objects")

matplotlib과 결합

Jupyter 노트북에서는 cv2.imshow가 동작하지 않는다. 대신 matplotlib으로 표시한다. 단, cv2의 BGR 이미지를 그대로 넘기면 색이 뒤집혀 보이므로 RGB로 변환해야 한다.

import cv2
import matplotlib.pyplot as plt

img = cv2.imread("sample.jpg")               # BGR
rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)   # 필수!

plt.figure(figsize=(8, 6))
plt.imshow(rgb)
plt.axis("off")
plt.title("Sample")
plt.show()

# 흑백
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
plt.imshow(gray, cmap="gray")
plt.show()

# 여러 단계 한 화면에
fig, axes = plt.subplots(1, 3, figsize=(15, 5))
axes[0].imshow(rgb);              axes[0].set_title("original")
axes[1].imshow(gray, cmap="gray"); axes[1].set_title("gray")
axes[2].imshow(cv2.Canny(gray, 100, 200), cmap="gray"); axes[2].set_title("edges")
for ax in axes: ax.axis("off")
plt.tight_layout()
plt.show()
Jupyter에서 cv2.imshow 금지 cv2.imshow는 GUI 백엔드가 필요해서 노트북에서는 커널이 멈춘다. 반드시 plt.imshow 또는 IPython.display.Image를 쓰자.

C++ OpenCV vs Python OpenCV

두 바인딩은 거의 동일한 함수 집합을 제공한다. 차이는 주로 속도와 개발 편의성이다.

항목C++ OpenCVPython OpenCV
API네이티브거의 동일 (cv2.X ↔ cv::X)
속도 (단일 함수)동일 (내부 구현 같음)동일
속도 (Python 루프)—느림 (벡터화 필요)
메모리 관리cv::Mat (참조 카운트)ndarray (자동)
배포실행 파일·DLL인터프리터 필요
적합 용도양산 통합·실시간프로토타이핑·검증

같은 함수 호출이 어떻게 닮았는지 보자.

// C++
#include <opencv2/opencv.hpp>
cv::Mat img = cv::imread("sample.jpg");
cv::Mat gray;
cv::cvtColor(img, gray, cv::COLOR_BGR2GRAY);
cv::Mat blur;
cv::GaussianBlur(gray, blur, cv::Size(5, 5), 1.0);
cv::imwrite("blur.png", blur);
# Python
import cv2
img = cv2.imread("sample.jpg")
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
blur = cv2.GaussianBlur(gray, (5, 5), 1.0)
cv2.imwrite("blur.png", blur)
현장 워크플로 Python으로 알고리즘 아이디어를 빠르게 검증한 뒤, 확정된 파이프라인을 C++로 포팅해 양산 장비에 통합한다. 대부분의 비전 함수는 1:1로 옮겨진다.

알고리즘 검증 스크립트

새 알고리즘을 양산 적용 전, 수백 장의 테스트 이미지로 PASS/FAIL 통계를 뽑는 일이 잦다.

import cv2, glob
from collections import Counter

results = Counter()
for path in glob.glob("test_images/*.bmp"):
    img = cv2.imread(path, cv2.IMREAD_GRAYSCALE)
    _, binary = cv2.threshold(img, 0, 255,
                              cv2.THRESH_BINARY + cv2.THRESH_OTSU)
    contours, _ = cv2.findContours(binary, cv2.RETR_EXTERNAL,
                                    cv2.CHAIN_APPROX_SIMPLE)
    verdict = "PASS" if 10 <= len(contours) <= 14 else "FAIL"
    results[verdict] += 1

print(results)   # Counter({'PASS': 187, 'FAIL': 13})

흔한 함정

BGR vs RGB 혼동 cv2는 BGR, Pillow·scikit-image·matplotlib은 RGB다. cv2로 읽은 이미지를 matplotlib에 그대로 넘기면 빨강과 파랑이 뒤집혀 보인다. 항상 cv2.cvtColor(img, cv2.COLOR_BGR2RGB)로 맞추자.
ndarray dtype 오버플로우 uint8 이미지에 + 50 같은 numpy 연산을 하면 255를 넘긴 값이 wrap-around된다 (250 + 50 = 44). 포화 연산을 원하면 cv2.add나 np.clip을 쓰자. 
import numpy as np, cv2

img = np.array([[250]], dtype=np.uint8)
print((img + 50)[0, 0])              # 44  (오버플로우!)
print(cv2.add(img, 50)[0, 0])        # 255 (포화 연산)
print(np.clip(img.astype(int) + 50, 0, 255)[0, 0])  # 255
Jupyter에서 cv2.imshow 사용 금지 커널이 멈춘다. matplotlib.pyplot.imshow 또는 IPython.display.Image를 사용한다. 단, matplotlib에 넘길 때 BGR→RGB 변환을 잊지 말자.
좌표계 헷갈림 cv2 함수는 (x, y) 튜플을, ndarray 슬라이싱은 [y, x] 순서를 쓴다. cv2.circle(img, (100, 50), ...)은 x=100, y=50이지만, img[100, 50]은 y=100, x=50이다.
Pillow의 좌표 vs OpenCV의 슬라이싱 Pillow crop((l, t, r, b))은 (왼쪽, 위, 오른쪽, 아래) 픽셀 좌표다. OpenCV 슬라이싱 img[t:b, l:r]과 같은 영역을 자른다.
한글 경로 문제 (Windows) cv2.imread("한글.jpg")가 None을 반환할 수 있다. np.fromfile + cv2.imdecode를 쓰면 안전하다. 
import cv2, numpy as np

def imread_unicode(path):
    data = np.fromfile(path, dtype=np.uint8)
    return cv2.imdecode(data, cv2.IMREAD_COLOR)

img = imread_unicode("샘플.jpg")
