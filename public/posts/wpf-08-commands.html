WPF 명령(Command)이란?

WPF 명령은 사용자의 행동(action)을 객체로 캡슐화하는 메커니즘이다. Button.Click 같은 이벤트 핸들러를 코드비하인드에 두면 View와 로직이 강하게 묶여 ViewModel 테스트가 어렵다. 명령을 쓰면 행동을 ViewModel 속성으로 노출하고 View는 그것을 바인딩만 하면 된다. MVVM 패턴의 핵심 부품이다.

이벤트 핸들러 방식과 명령 방식을 비교하면 다음과 같다.

<!-- 이벤트 방식 — 코드비하인드에 핸들러 필요 -->
<Button Content="저장" Click="SaveButton_Click"/>

<!-- 명령 방식 — ViewModel의 SaveCommand에 바인딩 -->
<Button Content="저장" Command="{Binding SaveCommand}"/>
// 이벤트 방식 — View.xaml.cs (코드비하인드)
private void SaveButton_Click(object sender, RoutedEventArgs e)
{
    var vm = (MainViewModel)DataContext;
    vm.Save();   // View가 ViewModel을 직접 호출 — 결합 발생
}

// 명령 방식 — MainViewModel.cs
public ICommand SaveCommand { get; }
// 생성자에서 RelayCommand로 초기화. View는 ViewModel을 모른다.
View → Binding → ICommand → ViewModel. View는 ViewModel의 구체 타입을 모른다.

ICommand 인터페이스

System.Windows.Input.ICommand는 세 멤버만 가진 단순한 인터페이스다. 모든 명령 객체가 구현해야 하는 계약이다.

namespace System.Windows.Input;

public interface ICommand
{
    // 명령이 실행 가능한지 반환. WPF는 이 결과로 컨트롤을 자동 enable/disable 한다.
    bool CanExecute(object? parameter);

    // 실제 명령 실행. CanExecute가 false여도 호출은 가능하다.
    void Execute(object? parameter);

    // 실행 가능 여부가 바뀌었음을 WPF에 알리는 이벤트.
    event EventHandler? CanExecuteChanged;
}
멤버역할호출 시점
CanExecute실행 가능 여부 질의WPF가 UI 갱신마다 폴링
Execute실제 행동 수행버튼 클릭, KeyGesture 발생 등
CanExecuteChangedWPF에 재질의 요청ViewModel 상태가 바뀐 직후 발화

CanExecute — UI 자동 비활성화

WPF의 강력한 점이다. Command가 바인딩된 Button·MenuItem은 CanExecute가 false면 자동으로 회색 처리되어 클릭이 막힌다. ViewModel은 별도의 IsEnabled 속성을 만들 필요가 없다.

public bool CanExecute(object? parameter)
{
    // 이미지가 로드되어 있을 때만 저장 가능.
    return m_currentImage != null && !m_isSaving;
}

// 상태가 바뀐 직후 — 반드시 CanExecuteChanged를 발화해야 UI가 갱신된다.
public Image? CurrentImage
{
    get => m_currentImage;
    set
    {
        m_currentImage = value;
        OnPropertyChanged();
        SaveCommand.RaiseCanExecuteChanged();   // ← 핵심
    }
}
CanExecuteChanged를 발화하지 않으면 버튼이 회색에서 깨어나지 않는다 상태가 바뀌었는데 CanExecuteChanged를 발화하지 않으면 WPF는 재질의하지 않는다. ViewModel에서 CanExecute에 영향을 주는 속성이 바뀔 때마다 잊지 말고 호출하자.

RoutedCommand 내장과 CommandBinding

WPF는 자주 쓰는 명령을 내장 RoutedCommand로 제공한다. ApplicationCommands.Save, Copy, Paste, Undo 등이다. 이 명령들은 단축키(Ctrl+S, Ctrl+C 등)와 입력 제스처가 미리 매핑되어 있다.

<Window x:Class="Imaging.MainWindow"
        xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation">
    <Window.CommandBindings>
        <!-- ApplicationCommands.Save에 핸들러를 연결 -->
        <CommandBinding Command="ApplicationCommands.Save"
                        Executed="SaveCommand_Executed"
                        CanExecute="SaveCommand_CanExecute"/>
    </Window.CommandBindings>

    <StackPanel>
        <!-- 같은 명령을 버튼과 메뉴 양쪽에 바인딩 가능 -->
        <Button Content="저장" Command="ApplicationCommands.Save"/>
        <MenuItem Header="저장(_S)" Command="ApplicationCommands.Save"/>
        <!-- Ctrl+S 단축키는 자동 지원 -->
    </StackPanel>
</Window>
// 코드비하인드 — RoutedCommand는 보통 코드비하인드에서 처리한다.
private void SaveCommand_Executed(object sender, ExecutedRoutedEventArgs e)
{
    // 이미지 저장 로직
}

private void SaveCommand_CanExecute(object sender, CanExecuteRoutedEventArgs e)
{
    e.CanExecute = m_currentImage != null;
}

RoutedCommand는 MVVM과 잘 안 맞는다 CommandBinding은 코드비하인드에 핸들러를 두기 때문에 ViewModel 단위 테스트가 어렵다. MVVM에서는 ViewModel에 ICommand 속성을 노출하는 방식(RelayCommand)이 표준이다.

RelayCommand 직접 구현

MVVM에서 가장 흔히 쓰는 패턴이다. Action·Func<bool>을 받아 Execute·CanExecute로 단순히 위임만 한다. 한 번 만들어두면 모든 ViewModel에서 재사용한다.

using System;
using System.Windows.Input;

namespace Imaging.Mvvm;

// Generic relay command — parameter type T.
public sealed class RelayCommand<T> : ICommand
{
    private readonly Action<T?> m_execute;
    private readonly Predicate<T?>? m_canExecute;

    public RelayCommand(Action<T?> execute, Predicate<T?>? canExecute = null)
    {
        m_execute = execute ?? throw new ArgumentNullException(nameof(execute));
        m_canExecute = canExecute;
    }

    public bool CanExecute(object? parameter)
        => m_canExecute == null || m_canExecute((T?)parameter);

    public void Execute(object? parameter)
        => m_execute((T?)parameter);

    public event EventHandler? CanExecuteChanged;

    // ViewModel이 호출하여 UI 갱신을 트리거한다.
    public void RaiseCanExecuteChanged()
        => CanExecuteChanged?.Invoke(this, EventArgs.Empty);
}

// Non-generic shortcut for parameter-less commands.
public sealed class RelayCommand : ICommand
{
    private readonly Action m_execute;
    private readonly Func<bool>? m_canExecute;

    public RelayCommand(Action execute, Func<bool>? canExecute = null)
    {
        m_execute = execute ?? throw new ArgumentNullException(nameof(execute));
        m_canExecute = canExecute;
    }

    public bool CanExecute(object? parameter) => m_canExecute?.Invoke() ?? true;
    public void Execute(object? parameter) => m_execute();

    public event EventHandler? CanExecuteChanged;
    public void RaiseCanExecuteChanged() => CanExecuteChanged?.Invoke(this, EventArgs.Empty);
}

사용 예제다. ViewModel 생성자에서 람다로 동작을 주입한다.

public class MainViewModel : ObservableObject
{
    private string m_inputPath = string.Empty;

    public string InputPath
    {
        get => m_inputPath;
        set
        {
            if (SetProperty(ref m_inputPath, value))

    public RelayCommand ClearCommand { get; }

            execute: () => LoadImage(m_inputPath),
            canExecute: () => !string.IsNullOrWhiteSpace(m_inputPath));

        ClearCommand = new RelayCommand(() => m_inputPath = string.Empty);
    }

    private void LoadImage(string path) { /* ... */ }
}

AsyncRelayCommand — async/await 결합

이미지 처리·파일 I/O·네트워크 요청은 비동기로 실행해야 UI가 멈추지 않는다. 그러나 ICommand.Execute는 void 반환이라 async void로 만들면 예외가 누수된다. AsyncRelayCommand는 Task를 반환하는 람다를 받아 안전하게 await 하고, 실행 중에는 자동으로 CanExecute를 false로 만들어 중복 실행을 막는다.

using System;
using System.Threading.Tasks;
using System.Windows.Input;

namespace Imaging.Mvvm;

public sealed class AsyncRelayCommand : ICommand
{
    private readonly Func<Task> m_executeAsync;
    private readonly Func<bool>? m_canExecute;
    private bool m_isRunning;

    public AsyncRelayCommand(Func<Task> executeAsync, Func<bool>? canExecute = null)
    {
        m_executeAsync = executeAsync ?? throw new ArgumentNullException(nameof(executeAsync));
        m_canExecute = canExecute;
    }

    public bool IsRunning
    {
        get => m_isRunning;
        private set
        {
            m_isRunning = value;
            RaiseCanExecuteChanged();
        }
    }

    public bool CanExecute(object? parameter)
        => !m_isRunning && (m_canExecute?.Invoke() ?? true);

    public async void Execute(object? parameter)
    {
        if (!CanExecute(parameter)) return;

        IsRunning = true;
        try
        {
            await m_executeAsync();   // 예외를 정상 캡처
        }
        catch (Exception ex)
        {
            // 로깅·에러 콜백 — async void로 예외 누수 방지.
            Logger.Error(ex, "AsyncRelayCommand failed");
        }
        finally
        {
            IsRunning = false;
        }
    }

    public event EventHandler? CanExecuteChanged;
    public void RaiseCanExecuteChanged() => CanExecuteChanged?.Invoke(this, EventArgs.Empty);
}
// 사용 예제 — 이미지 비동기 로드.
public AsyncRelayCommand LoadAsyncCommand { get; }

public MainViewModel()
{
    LoadAsyncCommand = new AsyncRelayCommand(async () =>
    {
        Status = "로딩 중...";
        CurrentImage = await Task.Run(() => Image.LoadFromFile(InputPath));
        Status = "완료";
    });
}
async void Execute의 예외는 UI 스레드를 폭파시킨다 async void에서 던진 예외는 호출자가 받지 못하고 SynchronizationContext로 올라가 앱이 종료될 수 있다. AsyncRelayCommand는 내부에서 try/catch로 잡아 로깅하거나 에러 콜백으로 전달해야 한다.

CommandParameter — XAML에서 매개변수 전달

같은 명령을 여러 버튼에서 재사용하면서 버튼마다 다른 인자를 넘기고 싶을 때 CommandParameter를 쓴다. 이 값이 그대로 Execute·CanExecute의 인자로 전달된다.

<StackPanel>
    <!-- 필터 종류를 매개변수로 전달 -->
    <Button Content="블러" Command="{Binding ApplyFilterCommand}"
            CommandParameter="Blur"/>
    <Button Content="샤픈" Command="{Binding ApplyFilterCommand}"
            CommandParameter="Sharpen"/>
    <Button Content="엣지" Command="{Binding ApplyFilterCommand}"
            CommandParameter="Edge"/>

    <!-- 선택된 아이템 자체를 전달 -->
    <ListBox x:Name="ImageList" ItemsSource="{Binding Images}"/>
    <Button Content="삭제"
            Command="{Binding DeleteCommand}"
            CommandParameter="{Binding SelectedItem, ElementName=ImageList}"/>
</StackPanel>
// ViewModel — 매개변수 타입을 RelayCommand<T>로 지정
public RelayCommand<string> ApplyFilterCommand { get; }
public RelayCommand<ImageItem> DeleteCommand { get; }

public MainViewModel()
{
    ApplyFilterCommand = new RelayCommand<string>(
        execute: kind => ApplyFilter(kind!),
        canExecute: kind => CurrentImage != null && !string.IsNullOrEmpty(kind));

    DeleteCommand = new RelayCommand<ImageItem>(
        execute: item => Images.Remove(item!),
        canExecute: item => item != null);
}

private void ApplyFilter(string kind)
{
    switch (kind)
    {
        case "Blur":    /* ... */ break;
        case "Sharpen": /* ... */ break;
        case "Edge":    /* ... */ break;
    }
}

CommandBinding vs Binding to Command

XAML에서 명령을 연결하는 두 가지 방식이 있다. 차이를 알면 상황에 맞게 고를 수 있다.

<!-- (A) CommandBinding 방식 — RoutedCommand + 코드비하인드 -->
<Window.CommandBindings>
    <CommandBinding Command="ApplicationCommands.Save"
                    Executed="OnSaveExecuted"/>
</Window.CommandBindings>
<Button Command="ApplicationCommands.Save"/>

<!-- (B) Binding to Command 방식 — ICommand + ViewModel -->
<Button Command="{Binding SaveCommand}"/>
비교(A) CommandBinding(B) Binding to Command
로직 위치코드비하인드ViewModel
명령 종류RoutedCommand임의의 ICommand
이벤트 라우팅지원 (Tunneling/Bubbling)없음 (단순 호출)
단위 테스트어려움쉬움
MVVM 적합도낮음높음
사용 시점전역 단축키, 표준 편집 명령대부분의 ViewModel 액션

CommunityToolkit.Mvvm — 현대적 방식

Microsoft가 공식 지원하는 CommunityToolkit.Mvvm NuGet 패키지는 소스 제너레이터로 RelayCommand·ObservableObject를 자동 생성한다. 보일러플레이트가 거의 사라진다.

<!-- vcpkg가 아닌 .NET NuGet 패키지 -->
<PackageReference Include="CommunityToolkit.Mvvm" Version="8.3.2" />
using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using System.Threading.Tasks;

// partial class 필수 — 소스 제너레이터가 같은 클래스에 코드를 채워 넣는다.
public partial class MainViewModel : ObservableObject
{
    // [ObservableProperty] — INotifyPropertyChanged 자동 구현.
    // m_currentImage 필드에서 CurrentImage 속성이 자동 생성된다.
    [ObservableProperty]
    [NotifyCanExecuteChangedFor(nameof(SaveCommand))]
    private Image? m_currentImage;

    [ObservableProperty]
    private string m_status = "대기 중";

    // [RelayCommand] — ICommand 속성이 메서드명에서 자동 생성.
    // SaveAsync 메서드 → SaveCommand 속성.
    [RelayCommand(CanExecute = nameof(CanSave))]
    private async Task SaveAsync()
    {
        Status = "저장 중...";
        await Task.Run(() => m_currentImage!.SaveToFile("out.png"));
        Status = "저장 완료";
    }

    private bool CanSave() => m_currentImage != null;
}
새 프로젝트에서는 CommunityToolkit.Mvvm을 우선 검토하자 Prism·MVVMLight를 대체하는 사실상 표준이다. async 명령·CanExecute 자동 갱신·INotifyPropertyChanged까지 모두 소스 제너레이터로 처리되어 ViewModel이 매우 간결해진다.

산업·이미지 적용 예제

비전 검사 ViewModel에 명령 패턴을 적용해본다. "검사 시작/정지", "이미지 저장" 같은 액션은 모두 ICommand로 노출한다. UI는 IsInspecting 상태에 따라 자동으로 버튼이 토글된다.

<Grid>
    <Grid.RowDefinitions>
        <RowDefinition Height="Auto"/>
        <RowDefinition Height="*"/>
    </Grid.RowDefinitions>

    <StackPanel Orientation="Horizontal" Margin="8">
        <Button Content="검사 시작" Width="100" Margin="4"
                Command="{Binding StartInspectionCommand}"/>
        <Button Content="검사 정지" Width="100" Margin="4"
                Command="{Binding StopInspectionCommand}"/>
        <Button Content="이미지 저장" Width="100" Margin="4"
                Command="{Binding SaveImageCommand}"
                CommandParameter="{Binding CurrentImage}"/>
        <TextBlock Text="{Binding Status}" Margin="12,8" VerticalAlignment="Center"/>
    </StackPanel>

    <Image Grid.Row="1" Source="{Binding CurrentImage.Preview}"/>
</Grid>
using System.Threading;
using System.Threading.Tasks;
using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;

public partial class InspectionViewModel : ObservableObject
{
    private readonly IInspectionService m_service;
    private CancellationTokenSource? m_cts;

    [ObservableProperty]
    [NotifyCanExecuteChangedFor(nameof(StartInspectionCommand))]
    [NotifyCanExecuteChangedFor(nameof(StopInspectionCommand))]
    private bool m_isInspecting;

    [ObservableProperty]
    [NotifyCanExecuteChangedFor(nameof(SaveImageCommand))]
    private ImageData? m_currentImage;

    [ObservableProperty]
    private string m_status = "대기 중";

    public InspectionViewModel(IInspectionService service)
    {
        m_service = service;
    }

    // 검사 시작 — 비동기, 실행 중에는 자동으로 비활성화.
    [RelayCommand(CanExecute = nameof(CanStart))]
    private async Task StartInspectionAsync()
    {
        IsInspecting = true;
        Status = "검사 중...";
        m_cts = new CancellationTokenSource();
        try
        {
            await foreach (var img in m_service.InspectAsync(m_cts.Token))
            {
                CurrentImage = img;
            }
            Status = "완료";
        }
        catch (OperationCanceledException)
        {
            Status = "정지됨";
        }
        finally
        {
            IsInspecting = false;
        }
    }
    private bool CanStart() => !IsInspecting;

    // 검사 정지.
    [RelayCommand(CanExecute = nameof(CanStop))]
    private void StopInspection() => m_cts?.Cancel();
    private bool CanStop() => IsInspecting;

    // 이미지 저장 — CommandParameter로 받은 이미지를 저장.
    [RelayCommand(CanExecute = nameof(CanSave))]
    private async Task SaveImageAsync(ImageData? image)
    {
        if (image == null) return;
        Status = "저장 중...";
        await image.SaveAsync($"capture_{DateTime.Now:yyyyMMdd_HHmmss}.png");
        Status = "저장 완료";
    }
    private bool CanSave(ImageData? image) => image != null && !IsInspecting;
}
실전 패턴 IsInspecting 같은 상태 플래그 하나에 NotifyCanExecuteChangedFor를 여러 개 붙여, 한 속성 변경으로 관련된 모든 명령이 갱신되도록 만든다. 검사 시작 시 정지·저장 버튼 상태가 자동으로 바뀐다.

흔한 함정

함정 1 — CanExecuteChanged 발화 누락

가장 흔한 버그다. CanExecute가 참조하는 속성을 바꿔놓고 RaiseCanExecuteChanged를 호출하지 않으면 버튼이 회색에서 깨어나지 않는다.

// 잘못된 예제
public string InputPath
{
    get => m_inputPath;
    set { m_inputPath = value; OnPropertyChanged(); }   // ← 빠진 호출
}

// 올바른 예제
public string InputPath
{
    get => m_inputPath;
    set
    {
        m_inputPath = value;
        OnPropertyChanged();
        LoadCommand.RaiseCanExecuteChanged();   // ← 반드시 호출
    }
}

함정 2 — async void Execute로 예외 누수

ICommand.Execute를 직접 async void로 구현하면 내부 예외가 호출자에게 전달되지 않고 SynchronizationContext로 올라가 앱이 죽는다. 반드시 try/catch로 감싸거나 AsyncRelayCommand를 사용하자.

// 위험한 예제 — 예외가 누수된다
public async void Execute(object? parameter)
{
    await DoWorkAsync();   // 여기서 예외가 나면 앱 종료 위험
}

// 안전한 예제
public async void Execute(object? parameter)
{
    try { await DoWorkAsync(); }
    catch (Exception ex) { Logger.Error(ex, "Command failed"); }
}

함정 3 — CommandManager.RequerySuggested 의존

예전 자료에서 자주 보이는 패턴이다. CanExecuteChanged를 CommandManager.RequerySuggested에 연결하면 WPF가 마우스·포커스 이벤트마다 자동 재질의한다. 편하지만 폴링 빈도가 높아 성능을 깎고, 트리거 시점도 모호하다.

// 비권장 — 폴링 기반, 시점 모호
public event EventHandler? CanExecuteChanged
{
    add { CommandManager.RequerySuggested += value; }
    remove { CommandManager.RequerySuggested -= value; }
}

// 권장 — 명시적 발화. 어디서 갱신되는지 코드로 보인다.
public event EventHandler? CanExecuteChanged;
public void RaiseCanExecuteChanged() => CanExecuteChanged?.Invoke(this, EventArgs.Empty);

함정 4 — RelayCommand 재발명

프로젝트마다 RelayCommand를 새로 짜는 사람이 많다. 2026년 기준 CommunityToolkit.Mvvm이 사실상 표준이므로 새 프로젝트라면 우선 검토하자. 직접 구현은 학습용·라이브러리 의존을 줄이고 싶을 때만 한다.
