왜 직접 커널을 짜는가

cv::cuda 모듈은 잘 만들어져 있고 대부분의 표준 연산을 커버한다. 그런데도 직접 커널을 짜야 하는 순간이 분명히 존재한다.

상황왜 직접 짜야 하는가
cv::cuda에 없는 연산커스텀 필터, 도메인 특화 알고리즘, 특수 비전 처리
여러 연산 합치기 (fusion)blur → threshold → morph을 한 커널로 합치면 메모리 왕복 제거
비표준 데이터 타입Mono10/12 packed, 비선형 인덱싱, 커스텀 구조체
메모리 재사용 최적화shared memory 타일링을 직접 제어해 글로벌 접근 최소화
제한된 입력 가정크기·정렬이 고정이면 분기를 제거하고 더 빠르게 가능
기본 원칙 먼저 cv::cuda로 만들고, 측정한 다음, 진짜 병목인 부분만 커널로 다시 짠다. 모든 걸 처음부터 직접 짜는 건 시간 낭비다.

이미지 커널의 공통 패턴

이미지 커널은 거의 같은 골격을 공유한다. 한 스레드가 한 픽셀을 맡고, 2D 인덱스를 계산하고, 경계를 체크하고, pitch로 줄 시작을 잡는다.

__global__ void ImageKernelTemplate(
    const uchar* __restrict__ src, size_t srcPitch,
    uchar* __restrict__ dst, size_t dstPitch,
    int width, int height)
{
    // 1) 픽셀 좌표 계산
    int x = blockIdx.x * blockDim.x + threadIdx.x;
    int y = blockIdx.y * blockDim.y + threadIdx.y;

    // 2) 경계 체크 (필수)
    if (x >= width || y >= height) return;

    // 3) pitch 기반 주소 계산 (cudaMallocPitch / GpuMat 호환)
    const uchar* srcRow = src + y * srcPitch;
    uchar*       dstRow = dst + y * dstPitch;

    // 4) 실제 처리
    dstRow[x] = srcRow[x];
}
이미지 커널의 표준 매핑 — Grid × Block × Thread 가 픽셀에 1:1 대응

pitch가 왜 중요한가

GPU는 행 시작 주소가 128/256바이트 정렬되어 있을 때 메모리 결합(coalescing)이 효율적이다. cudaMallocPitch와 cv::cuda::GpuMat은 각 행을 정렬하기 위해 행 끝에 padding을 둔다. 따라서 row * width가 아니라 row * pitch로 주소를 잡아야 한다.

표준 런치 설정

// 호스트 측 — 거의 모든 이미지 커널에서 재사용
inline void Launch2D(int width, int height, dim3& grid, dim3& block,
                     int bx = 16, int by = 16)
{
    block = dim3(bx, by);
    grid  = dim3((width  + bx - 1) / bx,
                 (height + by - 1) / by);
}

GaussianBlur 커널 — 분리 가능 필터

2D Gaussian은 1D Gaussian 두 번으로 분해할 수 있다(수평 ⊗ 수직). OpenCV 시리즈에서 다룬 그 분리 가능 필터다. 직접 짜면 K×K 곱셈을 2K로 줄여 거의 항상 빠르다.

방식곱셈 수 (픽셀당)K=11일 때
일반 2D 컨볼루션K²121 회
분리 가능(수평+수직)2K22 회

단순 2D 커널 (출발점)

__constant__ float c_kernel2D[25];  // 5x5

__global__ void Conv2D_5x5(
    const uchar* __restrict__ src, size_t srcPitch,
    uchar*       __restrict__ dst, size_t dstPitch,
    int width, int height)
{
    int x = blockIdx.x * blockDim.x + threadIdx.x;
    int y = blockIdx.y * blockDim.y + threadIdx.y;
    if (x >= width || y >= height) return;

    float sum = 0.0f;
    for (int ky = -2; ky <= 2; ++ky)
    {
        int sy = min(max(y + ky, 0), height - 1);  // BORDER_REPLICATE
        const uchar* row = src + sy * srcPitch;
        for (int kx = -2; kx <= 2; ++kx)
        {
            int sx = min(max(x + kx, 0), width - 1);
            sum += row[sx] * c_kernel2D[(ky + 2) * 5 + (kx + 2)];
        }
    }
    dst[y * dstPitch + x] = (uchar)__float2uint_rn(sum);
}

분리 가능 + Shared Memory 타일링 (수평 패스)

#define TILE_W 32
#define TILE_H 8
#define RADIUS 5

__constant__ float c_gauss1D[2 * RADIUS + 1];

__global__ void GaussianHorizontal(
    const uchar* __restrict__ src, size_t srcPitch,
    float*       __restrict__ tmp, size_t tmpPitch,
    int width, int height)
{
    __shared__ uchar tile[TILE_H][TILE_W + 2 * RADIUS];

    int tx = threadIdx.x;
    int ty = threadIdx.y;
    int x  = blockIdx.x * TILE_W + tx;
    int y  = blockIdx.y * TILE_H + ty;

    // shared로 행 단위 로드 (좌우 halo 포함)
    int baseX = blockIdx.x * TILE_W - RADIUS;
    for (int i = tx; i < TILE_W + 2 * RADIUS; i += TILE_W)
    {
        int sx = min(max(baseX + i, 0), width - 1);
        int sy = min(max(y, 0), height - 1);
        tile[ty][i] = src[sy * srcPitch + sx];
    }
    __syncthreads();

    if (x >= width || y >= height) return;

    float sum = 0.0f;
    #pragma unroll
    for (int k = -RADIUS; k <= RADIUS; ++k)
        sum += tile[ty][tx + RADIUS + k] * c_gauss1D[k + RADIUS];

    tmp[y * (tmpPitch / sizeof(float)) + x] = sum;
}

수직 패스도 같은 골격이다. tile을 세로로 잡고 ty 방향으로 halo를 두면 된다. 두 번 호출하면 2D Gaussian이 완성된다.

Constant memory가 필터 계수에 딱이다 모든 스레드가 같은 인덱스를 읽으면 __constant__는 broadcast로 한 번에 처리되어 매우 빠르다. 필터 커널 계수는 거의 항상 constant memory다.

Sobel 엣지 검출 커널

Sobel은 3×3 두 개의 필터로 Gx, Gy를 구하고 magnitude를 계산한다. 엣지 검출 글에서 다룬 그것이다. 3×3은 작아서 shared memory 없이도 잘 동작한다.

__global__ void SobelMagnitude(
    const uchar* __restrict__ src, size_t srcPitch,
    uchar*       __restrict__ dst, size_t dstPitch,
    int width, int height)
{
    int x = blockIdx.x * blockDim.x + threadIdx.x;
    int y = blockIdx.y * blockDim.y + threadIdx.y;
    if (x >= width || y >= height) return;

    // 경계 클램프
    int xm = max(x - 1, 0);
    int xp = min(x + 1, width  - 1);
    int ym = max(y - 1, 0);
    int yp = min(y + 1, height - 1);

    auto P = [&](int yy, int xx) -> int {
        return src[yy * srcPitch + xx];
    };

    // Gx
    int gx = -P(ym, xm) - 2 * P(y, xm) - P(yp, xm)
           +  P(ym, xp) + 2 * P(y, xp) + P(yp, xp);
    // Gy
    int gy = -P(ym, xm) - 2 * P(ym, x) - P(ym, xp)
           +  P(yp, xm) + 2 * P(yp, x) + P(yp, xp);

    // L1 근사 (sqrt보다 빠르고 실무에서 충분)
    int mag = abs(gx) + abs(gy);
    dst[y * dstPitch + x] = (uchar)min(mag, 255);
}
Sobel 커널의 메모리 접근 — 3×3 윈도우를 한 스레드가 모두 읽는다

임계화 커널 — 전역과 Otsu

임계화는 가장 단순한 픽셀 단위 연산이라 GPU에서 거의 메모리 대역폭에 묶인다.

전역 임계화

__global__ void ThresholdBinary(
    const uchar* __restrict__ src, size_t srcPitch,
    uchar*       __restrict__ dst, size_t dstPitch,
    int width, int height, uchar thresh, uchar maxVal)
{
    int x = blockIdx.x * blockDim.x + threadIdx.x;
    int y = blockIdx.y * blockDim.y + threadIdx.y;
    if (x >= width || y >= height) return;

    uchar v = src[y * srcPitch + x];
    dst[y * dstPitch + x] = (v > thresh) ? maxVal : 0;
}

Otsu 자동 임계값

Otsu는 히스토그램으로부터 클래스 간 분산이 최대가 되는 임계값을 찾는다. GPU에서는 (1) 히스토그램 계산, (2) 호스트에서 Otsu 식 평가, (3) 위 BinaryKernel 호출, 세 단계가 일반적이다. 256-bin 누적은 호스트에서 처리해도 무시할 만하다.

Histogram 커널 — atomic과 shared

히스토그램은 GPU 커널 중에서도 가장 까다로운 패턴이다. 수많은 스레드가 같은 256개의 bin에 동시에 쓰려고 하기 때문이다. shared histogram → global merge가 정석이다.

__global__ void Histogram256(
    const uchar* __restrict__ src, size_t srcPitch,
    int width, int height,
    unsigned int* __restrict__ hist)  // 256개
{
    __shared__ unsigned int sHist[256];

    // 1) shared 초기화 (256 스레드면 1회, 그 외엔 루프)
    int tid = threadIdx.y * blockDim.x + threadIdx.x;
    int blockSize = blockDim.x * blockDim.y;
    for (int i = tid; i < 256; i += blockSize)
        sHist[i] = 0;
    __syncthreads();

    // 2) 블록 내 픽셀들을 shared에 누적 (atomic이지만 충돌이 적다)
    int x = blockIdx.x * blockDim.x + threadIdx.x;
    int y = blockIdx.y * blockDim.y + threadIdx.y;
    if (x < width && y < height)
    {
        uchar v = src[y * srcPitch + x];
        atomicAdd(&sHist[v], 1u);
    }
    __syncthreads();

    // 3) global로 병합 (1 block당 256번)
    for (int i = tid; i < 256; i += blockSize)
        atomicAdd(&hist[i], sHist[i]);
}
왜 shared가 필수인가 global atomicAdd만 사용하면 수백만 스레드가 256 bin에 직접 충돌해서 사실상 직렬화된다. shared로 한 번 흡수하면 충돌이 같은 block 내부에 한정되어 수십~수백 배 빨라진다.

모폴로지 커널 — erosion과 dilation

3×3 구조 요소(structuring element)로 erosion/dilation을 직접 짠다. erosion은 윈도우의 min, dilation은 max다.

template <bool IsDilation>
__global__ void Morph3x3(
    const uchar* __restrict__ src, size_t srcPitch,
    uchar*       __restrict__ dst, size_t dstPitch,
    int width, int height)
{
    int x = blockIdx.x * blockDim.x + threadIdx.x;
    int y = blockIdx.y * blockDim.y + threadIdx.y;
    if (x >= width || y >= height) return;

    int xm = max(x - 1, 0), xp = min(x + 1, width  - 1);
    int ym = max(y - 1, 0), yp = min(y + 1, height - 1);

    uchar v = src[y * srcPitch + x];
    if (IsDilation)
    {
        v = max(v, src[ym * srcPitch + xm]);
        v = max(v, src[ym * srcPitch + x ]);
        v = max(v, src[ym * srcPitch + xp]);
        v = max(v, src[y  * srcPitch + xm]);
        v = max(v, src[y  * srcPitch + xp]);
        v = max(v, src[yp * srcPitch + xm]);
        v = max(v, src[yp * srcPitch + x ]);
        v = max(v, src[yp * srcPitch + xp]);
    }
    else  // erosion
    {
        v = min(v, src[ym * srcPitch + xm]);
        v = min(v, src[ym * srcPitch + x ]);
        v = min(v, src[ym * srcPitch + xp]);
        v = min(v, src[y  * srcPitch + xm]);
        v = min(v, src[y  * srcPitch + xp]);
        v = min(v, src[yp * srcPitch + xm]);
        v = min(v, src[yp * srcPitch + x ]);
        v = min(v, src[yp * srcPitch + xp]);
    }
    dst[y * dstPitch + x] = v;
}
모폴로지도 분리 가능하다 직사각형/십자형 구조 요소면 수평 erosion + 수직 erosion으로 분해 가능하다. 큰 커널일수록 효과가 크다.

Reduction 패턴 — 합·최대·최소

전체 픽셀의 평균이나 최대값 같은 단일 스칼라를 구하는 패턴이다. warp shuffle이 등장한 뒤로 reduction은 훨씬 간결해졌다.

__inline__ __device__ unsigned int warpReduceSum(unsigned int v)
{
    for (int off = 16; off > 0; off /= 2)
        v += __shfl_down_sync(0xffffffff, v, off);
    return v;
}

__global__ void ImageSum(
    const uchar* __restrict__ src, size_t srcPitch,
    int width, int height,
    unsigned long long* __restrict__ outSum)
{
    __shared__ unsigned int sPartial[32];  // 최대 32 warp/block

    int x = blockIdx.x * blockDim.x + threadIdx.x;
    int y = blockIdx.y * blockDim.y + threadIdx.y;

    unsigned int v = 0;
    if (x < width && y < height)
        v = src[y * srcPitch + x];

    // 1) warp 내부 reduce
    v = warpReduceSum(v);

    // 2) 각 warp의 lane 0이 shared에 저장
    int lane    = (threadIdx.y * blockDim.x + threadIdx.x) & 31;
    int warpId  = (threadIdx.y * blockDim.x + threadIdx.x) >> 5;
    if (lane == 0) sPartial[warpId] = v;
    __syncthreads();

    // 3) 첫 warp가 남은 부분합을 다시 reduce
    int nWarps = (blockDim.x * blockDim.y + 31) / 32;
    if (warpId == 0)
    {
        unsigned int x0 = (lane < nWarps) ? sPartial[lane] : 0u;
        x0 = warpReduceSum(x0);
        if (lane == 0) atomicAdd(outSum, (unsigned long long)x0);
    }
}

cv::cuda::GpuMat raw 포인터 연동

실무에서는 입력이 이미 cv::cuda::GpuMat로 들어오는 경우가 많다. GpuMat의 data와 step이 곧 raw 포인터와 pitch다.

#include <opencv2/core/cuda.hpp>

// 위에서 정의한 SobelMagnitude 호출
void RunSobelOnGpuMat(const cv::cuda::GpuMat& src, cv::cuda::GpuMat& dst,
                     cudaStream_t stream = 0)
{
    CV_Assert(src.type() == CV_8UC1);
    dst.create(src.size(), CV_8UC1);

    dim3 block(16, 16);
    dim3 grid((src.cols + 15) / 16, (src.rows + 15) / 16);

    SobelMagnitude<<<grid, block, 0, stream>>>(
        src.ptr<uchar>(), src.step,
        dst.ptr<uchar>(), dst.step,
        src.cols, src.rows);
    // 동기 확인은 디버그 빌드에서만
}

OpenCV의 cv::cuda::Stream과 합치고 싶으면 cv::cuda::StreamAccessor::getStream(stream)으로 cudaStream_t를 꺼내 쓰면 된다. 이러면 cv::cuda 호출과 직접 커널을 같은 스트림에 섞을 수 있다.

Fusion 예시 — Sobel + Threshold

__global__ void SobelThresholdFused(
    const uchar* __restrict__ src, size_t srcPitch,
    uchar*       __restrict__ dst, size_t dstPitch,
    int width, int height, int threshold)
{
    int x = blockIdx.x * blockDim.x + threadIdx.x;
    int y = blockIdx.y * blockDim.y + threadIdx.y;
    if (x >= width || y >= height) return;

    int xm = max(x - 1, 0), xp = min(x + 1, width  - 1);
    int ym = max(y - 1, 0), yp = min(y + 1, height - 1);

    int gx = -src[ym*srcPitch+xm] - 2*src[y*srcPitch+xm] - src[yp*srcPitch+xm]
           +  src[ym*srcPitch+xp] + 2*src[y*srcPitch+xp] + src[yp*srcPitch+xp];
    int gy = -src[ym*srcPitch+xm] - 2*src[ym*srcPitch+x] - src[ym*srcPitch+xp]
           +  src[yp*srcPitch+xm] + 2*src[yp*srcPitch+x] + src[yp*srcPitch+xp];

    int mag = abs(gx) + abs(gy);
    dst[y * dstPitch + x] = (mag > threshold) ? 255 : 0;  // 임계화까지 한 번에
}

산업 비전 관점 — Hot path를 커널로

검사 알고리즘은 보통 수십 개의 작은 연산이 줄줄이 이어진다. 모든 걸 직접 짤 필요는 없다. 다음 기준으로 골라낸다.

그대로 둘 곳직접 커널로 옮길 곳
cv::cuda 한 번 호출로 끝나는 큰 연산2~5개 작은 연산이 연속으로 묶이는 hot path
실행 시간 5% 미만인 부분전체의 30% 이상을 잡아먹는 단계
cv::cuda에 잘 매핑되는 표준 연산도메인 특화 알고리즘(특수 결함 검출, 비표준 디모자이크)

한 줄 카메라(line-scan) 라이브 검사는 latency가 곧 throughput이다. 한 프레임당 10 ms를 8 ms로 줄이는 게 25%의 라인 속도 향상이 된다. fusion만 잘해도 그 정도는 쉽게 줄어든다.

흔한 함정

경계 처리 누락 if (x >= width || y >= height) return;를 빠뜨리면 다음 행의 패딩 영역을 침범해 미세한 노이즈 또는 메모리 액세스 오류가 난다. compute-sanitizer로 잡힌다.
pitch 무시 y * width + x로 인덱싱하면 cv::cuda::GpuMat에서 한 행씩 어긋난다. 반드시 step 또는 pitch를 사용한다.
atomic 과다 사용 → 직렬화 global atomicAdd로 한 곳에 몰리면 GPU가 사실상 1개 SM처럼 동작한다. shared로 1차 흡수하고, 마지막 단계에서만 global atomic을 쓴다.
경계에서 divergence가 발생하는 if warp 안에서 일부 스레드만 if에 걸리면 분기 모두를 실행한다. 작은 분기는 무해하지만, 큰 분기는 min/max 클램프나 padding된 입력으로 회피한다.
측정하지 않은 자기만족 최적화 unroll, shared, constant를 다 박아 넣고 "빨라졌을 거다"라고 끝내는 게 가장 흔한 실수다. Nsight Compute로 메모리 throughput, occupancy, stall reason을 확인하기 전엔 끝난 게 아니다.
