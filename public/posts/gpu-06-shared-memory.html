Shared Memory 복습

2편 메모리 계층에서 다뤘듯이 GPU 메모리는 속도와 범위에 따라 계층을 이룬다. Shared memory는 같은 블록(block) 내 스레드들이 공유하는 매우 빠른 온칩(on-chip) 메모리다. 물리적으로는 SM(Streaming Multiprocessor) 내부의 SRAM이며, L1 캐시와 같은 영역을 나눠 쓴다.

메모리범위지연(cycle)용량/SM관리 주체
레지스터스레드~164KB컴파일러
Shared Memory블록~20–3048–164KB프로그래머
L1 캐시SM~30(공유)하드웨어
L2 캐시전체~2004–40MB하드웨어
Global Memory전체~400–800—프로그래머

Global memory에 비해 약 20–30배 빠르다. 같은 데이터를 여러 스레드가 반복 참조하면 shared memory에 한 번 적재한 뒤 재사용하는 것이 정석이다.

핵심 사용 시나리오 스레드들이 데이터를 공유하거나, 같은 데이터를 여러 번 읽거나, 인접 픽셀처럼 협력적 접근이 필요한 경우 shared memory를 사용한다.

선언과 사용

정적 선언

크기를 컴파일 타임에 고정하는 방식. 가장 단순하다.

__global__ void StaticSharedKernel(float* input, float* output, int n)
{
    // Compile-time fixed size shared memory.
    __shared__ float buf[256];

    int tid = threadIdx.x;
    int gid = blockIdx.x * blockDim.x + tid;

    // Load from global to shared.
    if (gid < n)
    {
        buf[tid] = input[gid];
    }
    __syncthreads();

    // Use shared data (here just write back).
    if (gid < n)
    {
        output[gid] = buf[tid] * 2.0f;
    }
}

// Launch.
StaticSharedKernel<<<gridDim, 256>>>(d_in, d_out, n);

동적 선언

커널 호출 시 크기를 지정하는 방식. extern 키워드로 선언하고 launch 시 세 번째 실행 인자로 바이트 단위 크기를 넘긴다.

__global__ void DynamicSharedKernel(float* input, float* output, int n)
{
    // Size is determined at launch time.
    extern __shared__ float buf[];

    int tid = threadIdx.x;
    int gid = blockIdx.x * blockDim.x + tid;

    if (gid < n)
    {
        buf[tid] = input[gid];
    }
    __syncthreads();

    if (gid < n)
    {
        output[gid] = buf[tid] * 2.0f;
    }
}

// The third <<<...>>> argument is shared memory bytes.
int blockSize = 256;
size_t sharedBytes = blockSize * sizeof(float);
DynamicSharedKernel<<<gridDim, blockSize, sharedBytes>>>(d_in, d_out, n);

동적 shared로 여러 배열 두기

동적 shared는 단일 버퍼만 선언 가능하다. 여러 배열이 필요하면 오프셋을 직접 계산한다.

__global__ void MultiArrayKernel(int n)
{
    extern __shared__ unsigned char smem[];

    // Manually split into multiple arrays.
    float* a = reinterpret_cast<float*>(smem);
    int*   b = reinterpret_cast<int*>(&a[blockDim.x]);

    int tid = threadIdx.x;
    a[tid] = 0.0f;
    b[tid] = 0;
    // ...
}

// Launch with total size for both arrays.
size_t bytes = blockSize * (sizeof(float) + sizeof(int));
MultiArrayKernel<<<grid, blockSize, bytes>>>(n);

__syncthreads()로 동기화

__syncthreads()는 같은 블록 내 모든 스레드가 이 지점에 도달할 때까지 대기하는 배리어(barrier)다. Shared memory에 쓴 데이터를 다른 스레드가 읽기 전에 반드시 호출해야 한다.

__global__ void ReverseInBlock(float* data)
{
    __shared__ float s[256];
    int tid = threadIdx.x;

    // Each thread writes its element.
    s[tid] = data[blockIdx.x * blockDim.x + tid];

    // Wait until all threads have finished writing.
    __syncthreads();

    // Now safe to read what other threads wrote.
    int reverseIdx = blockDim.x - 1 - tid;
    data[blockIdx.x * blockDim.x + tid] = s[reverseIdx];
}
__syncthreads() 누락은 데이터 race Shared memory에 쓰기를 한 직후 다른 스레드의 값을 읽는다면 동기화가 필수다. 빠뜨리면 비결정적(non-deterministic) 결과가 나오며, 디버그 빌드에서는 우연히 통과하다 릴리스에서 깨지는 일이 흔하다.

워프 내부 동기화

같은 워프(32 스레드) 내부라면 SIMT 특성상 lockstep으로 실행되지만, Volta 이후 architecture는 독립 스케줄링을 허용하므로 워프 내라도 __syncwarp()를 명시하는 것이 안전하다.

// Volta+ : warp-level barrier.
__syncwarp();              // entire warp
__syncwarp(0x0000FFFF);    // lower 16 lanes only (mask)

타일링 패턴 — 행렬 곱셈

Shared memory의 가장 대표적인 활용 패턴이 타일링(tiling)이다. 큰 데이터를 작은 타일 단위로 잘라 shared memory에 적재하고, 블록 내 스레드들이 협력해 재사용한다.

타일링 — 두 입력 행렬의 타일을 shared memory에 적재 후 블록 내 스레드가 재사용

타일링 없이 단순 구현하면 각 결과 원소마다 행 전체와 열 전체를 global memory에서 읽어야 한다. 타일링을 적용하면 같은 타일이 블록 내에서 여러 번 재사용되므로 global 접근 횟수가 TILE_SIZE배만큼 줄어든다.

#define TILE 16

__global__ void MatMulTiled(const float* A, const float* B, float* C, int N)
{
    __shared__ float tileA[TILE][TILE];
    __shared__ float tileB[TILE][TILE];

    int row = blockIdx.y * TILE + threadIdx.y;
    int col = blockIdx.x * TILE + threadIdx.x;

    float sum = 0.0f;

    // Iterate over tiles in the K dimension.
    for (int t = 0; t < N / TILE; ++t)
    {
        // Cooperative load: each thread loads one element.
        tileA[threadIdx.y][threadIdx.x] = A[row * N + (t * TILE + threadIdx.x)];
        tileB[threadIdx.y][threadIdx.x] = B[(t * TILE + threadIdx.y) * N + col];

        __syncthreads();    // wait until tile is fully loaded

        // Compute partial dot product using shared tiles.
        #pragma unroll
        for (int k = 0; k < TILE; ++k)
        {
            sum += tileA[threadIdx.y][k] * tileB[k][threadIdx.x];
        }

        __syncthreads();    // wait before overwriting tiles in next iteration
    }

    C[row * N + col] = sum;
}

// Launch.
dim3 block(TILE, TILE);
dim3 grid(N / TILE, N / TILE);
MatMulTiled<<<grid, block>>>(dA, dB, dC, N);
왜 __syncthreads()가 두 번 필요한가 첫 번째는 타일 로드 완료 후 모든 스레드가 같이 읽기 위해서, 두 번째는 다음 반복에서 새 타일로 덮어쓰기 전에 모든 스레드의 계산이 끝났음을 보장하기 위해서다.

이미지 컨볼루션 예제

이미지 처리에서 3×3, 5×5 같은 작은 커널을 이미지 전체에 적용하는 컨볼루션도 타일링의 좋은 사례다. 다만 경계에서 이웃 픽셀을 참조해야 하므로 halo(이웃 여유 영역)를 포함해 적재한다.

컨볼루션 — 블록 크기 16×16에 halo 1픽셀씩 더해 18×18을 shared로 로드
#define BLOCK 16
#define RADIUS 1
#define TILE_W (BLOCK + 2 * RADIUS)   // 18

__constant__ float c_kernel[9] = {
    1/16.f, 2/16.f, 1/16.f,
    2/16.f, 4/16.f, 2/16.f,
    1/16.f, 2/16.f, 1/16.f
};

__global__ void GaussianBlur3x3(const unsigned char* input,
                                unsigned char* output,
                                int width, int height)
{
    __shared__ unsigned char tile[TILE_W][TILE_W];

    int tx = threadIdx.x;
    int ty = threadIdx.y;

    // Global coordinates of this thread's output pixel.
    int gx = blockIdx.x * BLOCK + tx;
    int gy = blockIdx.y * BLOCK + ty;

    // Coordinates inside the shared tile (offset by halo radius).
    int lx = tx + RADIUS;
    int ly = ty + RADIUS;

    // Helper: clamped global read.
    auto loadPixel = [&](int x, int y) -> unsigned char {
        x = max(0, min(x, width - 1));
        y = max(0, min(y, height - 1));
        return input[y * width + x];
    };

    // 1) Each thread loads its own center pixel.
    tile[ly][lx] = loadPixel(gx, gy);

    // 2) Border threads also load halo pixels.
    if (tx < RADIUS)
    {
        tile[ly][lx - RADIUS]       = loadPixel(gx - RADIUS, gy);
        tile[ly][lx + BLOCK]        = loadPixel(gx + BLOCK, gy);
    }
    if (ty < RADIUS)
    {
        tile[ly - RADIUS][lx]       = loadPixel(gx, gy - RADIUS);
        tile[ly + BLOCK][lx]        = loadPixel(gx, gy + BLOCK);
    }
    // (corners omitted for brevity — load similarly)

    __syncthreads();

    if (gx >= width || gy >= height) return;

    // 3) Apply 3x3 convolution from shared memory.
    float sum = 0.0f;
    #pragma unroll
    for (int ky = -RADIUS; ky <= RADIUS; ++ky)
    {
        #pragma unroll
        for (int kx = -RADIUS; kx <= RADIUS; ++kx)
        {
            sum += tile[ly + ky][lx + kx]
                 * c_kernel[(ky + RADIUS) * 3 + (kx + RADIUS)];
        }
    }

    output[gy * width + gx] = static_cast<unsigned char>(sum);
}
왜 halo가 필요한가 16×16 출력을 만들 때 3×3 커널을 적용하려면 각 경계 픽셀이 바로 옆 이웃을 읽어야 한다. 이웃까지 shared로 미리 적재해 두면 컨볼루션 루프 안에서 추가 global 접근이 사라진다.

뱅크 충돌(Bank Conflict)

Shared memory는 빠른 병렬 접근을 위해 32개의 뱅크(bank)로 나뉘어 있다. 워프(32 스레드)가 한 번에 메모리에 접근할 때, 32 스레드가 서로 다른 뱅크에 접근하면 1 cycle에 모두 처리된다. 하지만 두 스레드가 같은 뱅크의 다른 주소를 동시에 요청하면 직렬화되어 성능이 떨어진다. 이를 뱅크 충돌이라고 한다.

3가지 패턴 — 서로 다른 뱅크(이상), 같은 뱅크 다른 주소(충돌), 같은 주소(브로드캐스트)

현대 GPU에서 shared memory는 4바이트 워드 단위로 32개 뱅크에 순차 매핑된다. 즉 float buf[N]에서 buf[i]는 뱅크 i % 32에 위치한다.

접근 패턴설명성능
buf[tid]스레드 i → 뱅크 i. 완벽한 분산1 cycle (최적)
buf[tid * 2]홀짝만 사용. 2-way 충돌2 cycles
buf[tid * 32]모두 뱅크 0. 32-way 충돌32 cycles (최악)
buf[0] (모든 스레드)같은 주소 동시 읽기1 cycle (broadcast)
buf[tid * 4]뱅크 0, 4, 8, ... 8-way 충돌4 cycles
뱅크 충돌은 조용한 성능 킬러 결과 자체는 정확하므로 기능 테스트는 통과한다. 하지만 32-way 충돌은 최적 대비 32배 느리게 동작한다. Nsight Compute의 shared_load_bank_conflicts 지표로 확인해야 한다.

뱅크 충돌 회피 기법

패딩(Padding)

2D shared 배열에서 한 열을 통째로 접근할 때(buf[i][tid]처럼 i가 변하는 접근)는 모든 스레드가 같은 뱅크에 모일 수 있다. 한 열에 더미 칸 하나를 추가하면 stride가 어긋나 충돌이 사라진다.

// 충돌 발생 — 32 스레드가 같은 열 접근 시 모두 같은 뱅크
__shared__ float buf[32][32];
// buf[0][tid], buf[1][tid], ... 모두 tid%32 뱅크

// 해결 — 한 칸 패딩
__shared__ float buf[32][33];   // +1 padding
// 이제 buf[i][tid]의 뱅크 = (i * 33 + tid) % 32
// i가 바뀌어도 뱅크가 분산됨

전치(transpose) 커널 예시

행렬 전치는 뱅크 충돌이 발생하기 쉬운 대표 예시다. 패딩 한 줄만 추가해도 큰 성능 향상을 얻는다.

#define TILE 32

__global__ void TransposeNaive(const float* in, float* out, int N)
{
    __shared__ float tile[TILE][TILE];     // bank conflict on column read!

    int x = blockIdx.x * TILE + threadIdx.x;
    int y = blockIdx.y * TILE + threadIdx.y;

    tile[threadIdx.y][threadIdx.x] = in[y * N + x];
    __syncthreads();

    int xt = blockIdx.y * TILE + threadIdx.x;
    int yt = blockIdx.x * TILE + threadIdx.y;
    out[yt * N + xt] = tile[threadIdx.x][threadIdx.y];   // column access → conflict
}

__global__ void TransposePadded(const float* in, float* out, int N)
{
    __shared__ float tile[TILE][TILE + 1]; // +1 padding removes conflict

    int x = blockIdx.x * TILE + threadIdx.x;
    int y = blockIdx.y * TILE + threadIdx.y;

    tile[threadIdx.y][threadIdx.x] = in[y * N + x];
    __syncthreads();

    int xt = blockIdx.y * TILE + threadIdx.x;
    int yt = blockIdx.x * TILE + threadIdx.y;
    out[yt * N + xt] = tile[threadIdx.x][threadIdx.y];   // no conflict
}

Stride 조정과 접근 패턴 재설계

// 나쁜 접근 — stride 32, 모두 뱅크 0
for (int i = 0; i < n; i += 32) sum += buf[tid + i];

// 좋은 접근 — 연속 인덱스
for (int i = 0; i < n; ++i) sum += buf[tid + i * blockDim.x];

// Reduction의 경우 stride를 워프 폭으로 자연스럽게 분산
for (int s = blockDim.x / 2; s > 0; s >>= 1)
{
    if (tid < s) buf[tid] += buf[tid + s];
    __syncthreads();
}

선언__shared__ float buf[256];extern __shared__ float buf[];
크기 결정컴파일 타임런타임(커널 launch 시)
여러 배열가능, 자연스러움오프셋 수동 계산 필요
launch 인자3번째 자리 생략 가능3번째 자리에 바이트 지정
유연성낮음높음
주 사용처크기가 고정된 알고리즘크기를 호출자가 결정
혼합 사용도 가능 정적 선언으로 고정 크기 버퍼를, 동적으로 가변 크기 버퍼를 동시에 사용할 수 있다. 동적 부분은 launch 시 바이트 크기만 넘기면 된다.

Shared Memory와 L1 비율 조정

SM 내부의 통합 SRAM은 shared memory와 L1 캐시가 나눠 쓴다. 워크로드 특성에 따라 비율을 조정할 수 있다.

// 커널별 캐시 선호도 설정.
cudaFuncSetCacheConfig(MyKernel, cudaFuncCachePreferShared);   // shared 우선
cudaFuncSetCacheConfig(MyKernel, cudaFuncCachePreferL1);       // L1 우선
cudaFuncSetCacheConfig(MyKernel, cudaFuncCachePreferEqual);    // 동등
cudaFuncSetCacheConfig(MyKernel, cudaFuncCachePreferNone);     // 기본

// Volta 이후: opt-in으로 동적 shared memory 한도 상향.
cudaFuncSetAttribute(MyKernel,
    cudaFuncAttributeMaxDynamicSharedMemorySize,
    96 * 1024);  // 96KB

MyKernel<<<grid, block, 96 * 1024>>>(...);
아키텍처별 한도 차이 기본 한도(예: 48KB)를 넘는 동적 shared를 쓰려면 cudaFuncAttributeMaxDynamicSharedMemorySize 설정이 필수다. Ampere(A100)는 최대 164KB까지 지원한다.

흔한 함정

함정 1 — __syncthreads() 누락

// 잘못된 코드
__shared__ float s[256];
s[threadIdx.x] = input[gid];
// __syncthreads() 누락!
float v = s[(threadIdx.x + 1) % 256];   // 다른 스레드의 값이 아직 안 쓰였을 수 있음

함정 2 — 발산(divergent) 코드 내 동기화

// 데드락 위험 — if 안의 __syncthreads()
if (threadIdx.x < 32)
{
    s[threadIdx.x] = ...;
    __syncthreads();        // ★ 위험: 다른 스레드는 도달 못 함
}

// 올바른 패턴 — 모든 스레드가 도달
s[threadIdx.x] = ...;       // 모든 스레드 실행
__syncthreads();            // 블록 전체가 도달
if (threadIdx.x < 32)
{
    // ...
}
__syncthreads()는 반드시 모든 스레드가 도달해야 한다 조건문이나 early return으로 일부 스레드만 동기화 지점에 도달하면 영구 데드락에 빠진다. 블록 전체가 도달 가능한 위치에만 두자.

함정 3 — 뱅크 충돌 미인지

코드는 정확히 동작하지만 성능이 기대치에 못 미친다면 첫 번째로 의심할 것이 뱅크 충돌이다. Nsight Compute로 측정한다.

$ ncu --metrics l1tex__data_bank_conflicts_pipe_lsu_mem_shared.sum ./my_app
# shared_load_bank_conflicts / shared_store_bank_conflicts 지표 확인

함정 4 — Shared memory 크기 초과

// 컴파일 에러 또는 런타임 launch 실패
__shared__ float huge[100 * 1024];   // 400KB - SM 한도 초과

// 런타임 확인
cudaDeviceProp prop;
cudaGetDeviceProperties(&prop, 0);
printf("Shared per block: %zu bytes\n", prop.sharedMemPerBlock);
printf("Shared per SM   : %zu bytes\n", prop.sharedMemPerMultiprocessor);
Occupancy와의 트레이드오프 블록당 shared를 많이 쓰면 SM에 동시 실행 가능한 블록 수가 줄어 occupancy가 떨어진다. 무조건 많이 쓰는 게 능사가 아니다. cudaOccupancyMaxActiveBlocksPerMultiprocessor로 확인하자.

함정 5 — Volta 이후 워프 독립 스케줄링

Volta 이전에는 워프 내부 스레드가 lockstep으로 진행되어 워프 내 동기화가 암묵적이었다. Volta(SM 7.0) 이후는 독립 스케줄링이 가능해 워프 내 통신에도 __syncwarp()를 명시해야 안전하다.

// Pre-Volta 가정 — Volta+에서 깨질 수 있음
if (tid < 32)
{
    s[tid] += s[tid + 16];   // 워프 내 — 이전에는 동기화 불필요
    s[tid] += s[tid + 8];
    // ...
}

// Volta+ 안전한 패턴
if (tid < 32)
{
    s[tid] += s[tid + 16];
    __syncwarp();
    s[tid] += s[tid + 8];
    __syncwarp();
    // ...
}
